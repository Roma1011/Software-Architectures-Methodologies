
პირველი სპრინტის დაგეგმვის შეხვედრაზე, გუნდი, ოთხივე, ბექლოგში უკვე შეფასებულ story-ს იხილავს — ოთახის შექმნა, ოთახების ჩამონათვალის ნახვა, ოთახში შეტყობინებების ნახვა, და ჩვეულებრივი ტექსტის გაგზავნა (ჯამში, 9 ქულა). სანამ, კონკრეტულ, ცალკეულ story-ებზე გადავა, გუნდი, ჯერ, რამდენიმე, **არქიტექტურულ გადაწყვეტილებას** იღებს, რომელიც, მთელ სპრინტს შეეხება.

### არქიტექტურული ჩონჩხი

გუნდი ირჩევს, უკვე ნაცნობ, სამფენიან სტრუქტურას — **კონტროლერი** (შემომავალი მოთხოვნის მიღება), **სერვისი** (ბიზნეს ლოგიკა), **საცავი** (მდგრადობა) — თითოეული, საკუთარი ინტერფეისის მიღმა, Stairway-ის სტილში, დამოკიდებულების ინექციით შეკავშირებული. ვინაიდან, სპრინტ ნულში, გუნდმა უკვე გადაწყვიტა, ვებ-ჩარჩოდ, ორგანიზაციაში უკვე ცნობილი ტექნოლოგია აერჩია, ხოლო მდგრადობისთვის, თავდაპირველად, მარტივი, **მეხსიერებაში დაფუძნებული საცავი** გამოეყენებინა — ნამდვილ ბაზასთან ინტეგრაცია, შემდეგ სპრინტში, ინტერფეისის შეცვლის გარეშე, დაემატება.

```csharp
public interface IRoomRepository
{
    Room Create(string name, string ownerName);
    Room GetById(Guid id);
    IEnumerable<Room> GetAll();
}

public interface IMessageRepository
{
    Message Add(Guid roomId, string authorName, string text);
    IEnumerable<Message> GetByRoom(Guid roomId);
}
```

### Story 1: ოთახის შექმნა

`RoomService`, ერთადერთ პასუხისმგებლობას ინარჩუნებს — მოთხოვნის ვალიდაცია და საცავზე დელეგირება — ხოლო კონტროლერი, მხოლოდ HTTP-სპეციფიკურ დეტალებზეა პასუხისმგებელი:

```csharp
public interface IRoomService
{
    RoomDto CreateRoom(string name, string ownerName);
    IEnumerable<RoomDto> ListRooms();
}

public class RoomService : IRoomService
{
    private readonly IRoomRepository repository;
    public RoomService(IRoomRepository repository) => this.repository = repository;

    public RoomDto CreateRoom(string name, string ownerName)
    {
        if (string.IsNullOrWhiteSpace(name))
            throw new ArgumentException("ოთახის სახელი ცარიელი ვერ იქნება", nameof(name));
        if (string.IsNullOrWhiteSpace(ownerName))
            throw new ArgumentException("მფლობელის სახელი აუცილებელია", nameof(ownerName));

        var room = repository.Create(name, ownerName);
        return new RoomDto(room.Id, room.Name, room.OwnerName);
    }

    public IEnumerable<RoomDto> ListRooms() =>
        repository.GetAll().Select(r => new RoomDto(r.Id, r.Name, r.OwnerName));
}

[ApiController, Route("api/rooms")]
public class RoomsController : ControllerBase
{
    private readonly IRoomService roomService;
    public RoomsController(IRoomService roomService) => this.roomService = roomService;

    [HttpPost]
    public ActionResult<RoomDto> Create([FromBody] CreateRoomRequest request) =>
        Ok(roomService.CreateRoom(request.Name, request.OwnerName));

    [HttpGet]
    public ActionResult<IEnumerable<RoomDto>> List() => Ok(roomService.ListRooms());
}
```

ვინაიდან `RoomService`, `IRoomRepository`-ზეა დამოკიდებული — და არა კონკრეტულ, მაგალითად, მეხსიერებაზე დაფუძნებულ იმპლემენტაციაზე — მისი ერთეულოვანი ტესტირება, საცავის „მოტყუებით" (mock), მარტივია:

```csharp
[TestMethod]
public void CreatingRoomWithEmptyNameThrows()
{
    var service = new RoomService(Mock.Of<IRoomRepository>());
    Assert.ThrowsException<ArgumentException>(() => service.CreateRoom("", "Alice"));
}

[TestMethod]
public void CreatingRoomDelegatesToRepository()
{
    var repositoryMock = new Mock<IRoomRepository>();
    repositoryMock.Setup(r => r.Create("გუნდის სამზარეულო", "Alice"))
        .Returns(new Room(Guid.NewGuid(), "გუნდის სამზარეულო", "Alice"));

    var service = new RoomService(repositoryMock.Object);
    var result = service.CreateRoom("გუნდის სამზარეულო", "Alice");

    Assert.AreEqual("გუნდის სამზარეულო", result.Name);
}
```

### Story 2: ოთახების ჩამონათვალის ნახვა

ეს story, პრაქტიკულად, უკვე მზადაა — `RoomService.ListRooms()`-მა, ეს საჭიროება, ერთი, დამატებითი მეთოდით დაფარა (იხილეთ ზემოთ). ცალკე გამორჩეული, არქიტექტურული საკითხი აქ ისაა, რომ, გუნდი, შეგნებულად ტოვებს, **დახარისხების** საკითხს (უახლესი, თუ ანბანური რიგით?) — ვინაიდან, კლიენტმა, ამის შესახებ, ჯერ არაფერი უთქვამს, და, საჭიროებაზე მეტი გადაწყვეტილების ნაადრევად მიღება, დაცული ვარიაციის პრინციპს (protected variation) ეწინააღმდეგება.

### Story 3: ოთახში შეტყობინებების ნახვა

„შეტყობინებების ნახვის" story, ცალკე, `IMessageService`-ის იმპლემენტაციას საჭიროებს — მნიშვნელოვანი დეტალით: **შეტყობინება, კონკრეტულ ოთახს უნდა ეკუთვნოდეს**, ანუ, `MessageService`, თავად, `IRoomRepository`-ზეც დამოკიდებულია, რომ დარწმუნდეს, მითითებული ოთახი, ნამდვილად არსებობს:

```csharp
public interface IMessageService
{
    MessageDto Send(Guid roomId, string authorName, string text);
    IEnumerable<MessageDto> GetHistory(Guid roomId);
}

public class MessageService : IMessageService
{
    private readonly IMessageRepository messages;
    private readonly IRoomRepository rooms;

    public MessageService(IMessageRepository messages, IRoomRepository rooms)
    {
        this.messages = messages;
        this.rooms = rooms;
    }

    public IEnumerable<MessageDto> GetHistory(Guid roomId)
    {
        if (rooms.GetById(roomId) == null)
            throw new InvalidOperationException("მითითებული ოთახი არ არსებობს");

        return messages.GetByRoom(roomId)
            .OrderBy(m => m.SentAt)
            .Select(m => new MessageDto(m.Id, m.AuthorName, m.Text, m.SentAt));
    }

    public MessageDto Send(Guid roomId, string authorName, string text)
    {
        if (rooms.GetById(roomId) == null)
            throw new InvalidOperationException("მითითებული ოთახი არ არსებობს");
        if (string.IsNullOrWhiteSpace(text))
            throw new ArgumentException("ცარიელი შეტყობინების გაგზავნა შეუძლებელია", nameof(text));

        var message = messages.Add(roomId, authorName, text);
        return new MessageDto(message.Id, message.AuthorName, message.Text, message.SentAt);
    }
}
```

### Story 4: ჩვეულებრივი ტექსტის გაგზავნა

ეს story, ზემოთ უკვე ნაწილობრივ დაიფარა (`MessageService.Send`) — თუმცა, დაგეგმვის შეხვედრაზე, ეს ის story იყო, რომლის საწყისი შეფასებაც, გუნდის შიგნით, ყველაზე მეტად განსხვავდებოდა. თავდაპირველად, ერთ დეველოპერს ეგონა, რომ, ეს — მხოლოდ, ტექსტის ეკრანზე გამოტანაა, და მხოლოდ დისკუსიის შემდეგ გაცხადდა, რომ, შეტყობინება, **შენახული** და, **სხვა წევრებისთვისაც ხილვადი** უნდა იყოს — რაც, თავად, `IMessageRepository`-ისა და `IMessageService`-ის საჭიროებას განაპირობა. ეს კარგად აჩვენებს, თუ რატომაა, story-ის განხილვის ეტაპი, ხშირად, თავად კოდის წერაზე არანაკლებ ღირებული.

### Story-ების ურთიერთდამოკიდებულება და კომპოზიციის ფესვი

ოთხივე story, საბოლოო ჯამში, ერთი, საერთო კომპოზიციის ფესვში იკვრება:

```csharp
var services = new ServiceCollection();
services.AddSingleton<IRoomRepository, InMemoryRoomRepository>();
services.AddSingleton<IMessageRepository, InMemoryMessageRepository>();
services.AddScoped<IRoomService, RoomService>();
services.AddScoped<IMessageService, MessageService>();
```

ყურადღება გავამახვილოთ იმაზეც, რომ საცავები, **singleton**-ის სახითაა რეგისტრირებული (რომ მონაცემი, მოთხოვნებს შორის შენარჩუნდეს, თუნდაც, ჯერ, მხოლოდ მეხსიერებაშივე), ხოლო სერვისები — **scoped**-ის სახით, თითოეული, ცალკეული HTTP მოთხოვნისთვის.

### სპრინტის დემონსტრირება

სპრინტის ბოლოს, გუნდი, ოთხივე story-ს, რეალურ, გაშვებულ გარემოში აჩვენებს — ორი, ერთდროულად „დამსწრე" ბრაუზერის ფანჯარა, ერთი, სხვისი მიერ შექმნილი ოთახის მაგალითზე, აჩვენებს, თუ როგორ ხედავს ორივე მომხმარებელი, ერთმანეთის, რეალურ დროში გაგზავნილ ტექსტურ შეტყობინებებს. სპრინტის მიზანი — მიღწეულია.

### სპრინტის რეტროსპექტივა

გუნდი, სტანდარტულ ხუთ კითხვას სვამს:

- **რა წავიდა კარგად?** — ინტერფეისებზე დაფუძნებულმა დიზაინმა, საშუალება მისცა, ოთხივე story, ერთმანეთისგან, თითქმის დამოუკიდებლად ემუშავათ, ვინაიდან, თითოეული ინტერფეისის „კონტრაქტი", ადრეულ ეტაპზევე იყო შეთანხმებული.
- **რა გამოწვევა იყო?** — ახალბედა დეველოპერმა, თავდაპირველად, `IMessageService`-ს, პირდაპირ, `RoomsController`-შიც სცადა გამოეყენებინა („ერთი კონტროლერი, ორივეს გააკეთებდა") — გუნდმა, ეს, SRP-ის დარღვევად ჩათვალა, და, ორ, ცალკეულ კონტროლერად გაყო.
- **რა უნდა დავიწყოთ?** — ტესტ-ანალიტიკოსმა, შესთავაზა, თითოეული, ახალი story-ს დაწყებამდე, მისი მისაღები კრიტერიუმები (acceptance criteria), წერილობით ფორმდებოდეს — არა მხოლოდ, ზეპირი შეთანხმებით.
- **რა უნდა შევწყვიტოთ?** — ერთხელ, სპრინტის შუაში, გუნდმა, დაუგეგმავად, `IMessageRepository`-ის ინტერფეისი შეცვალა, უკვე დაწერილი ტესტების გათვალისწინების გარეშე — რამაც, დროებით, სამი ტესტი ჩაშალა. გუნდმა გადაწყვიტა, სამომავლოდ, ინტერფეისის ცვლილება, ყოველთვის, ჯერ, დანარჩენ გუნდთან შეთანხმდეს.
- **რა შევინარჩუნოთ?** — Planning poker-ის სესია, საკმაოდ სწრაფად და ეფექტურად ჩატარდა, ვინაიდან, story-ები, უკვე, საკმარისად მცირე და კონკრეტული იყო — ეს პრაქტიკა, შემდეგ სპრინტშიც გაგრძელდება.
